cmake_minimum_required(VERSION 3.24)
project(combadge_speaker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(ORT_INCLUDE_DIR "" CACHE PATH "ONNX Runtime include/onnxruntime/core/session directory")
set(ORT_LIBRARY "" CACHE FILEPATH "ONNX Runtime shared library")
if(NOT EXISTS "${ORT_INCLUDE_DIR}/onnxruntime_cxx_api.h" OR NOT EXISTS "${ORT_LIBRARY}")
  message(FATAL_ERROR "Set ORT_INCLUDE_DIR and ORT_LIBRARY to the QNX ONNX Runtime build")
endif()
include(FetchContent)
set(KALDI_NATIVE_FBANK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(KALDI_NATIVE_FBANK_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(knf
  GIT_REPOSITORY https://github.com/csukuangfj/kaldi-native-fbank.git
  GIT_TAG b09e686fe2084732ddd30d1ef80acfc0f13eaf01)
FetchContent_MakeAvailable(knf)
add_executable(speaker-worker worker.cc)
target_include_directories(speaker-worker PRIVATE ${ORT_INCLUDE_DIR} ${knf_SOURCE_DIR})
target_link_libraries(speaker-worker PRIVATE kaldi-native-fbank-core ${ORT_LIBRARY})
install(TARGETS speaker-worker RUNTIME DESTINATION bin)
